```{r}
fmt_bin_pct <- function(x) {
  n <- sum(x == 1, na.rm = TRUE)
  pct <- 100 * mean(x == 1, na.rm = TRUE)

  sprintf("%s (%.1f%%)", format(n, big.mark = ","), pct)
}

fmt_p <- function(p) {
  if (is.na(p)) return("")
  if (p < .001) return("<0.001")
  sprintf("%.3f", p)
}

make_bin_row <- function(label, var) {

  overall <- model_df[[var]]

  app <- model_df %>%
    filter(appalachia == 1) %>%
    pull(.data[[var]])

  non <- model_df %>%
    filter(appalachia == 0) %>%
    pull(.data[[var]])

  pval <- suppressWarnings(
    tryCatch(
      chisq.test(table(model_df$appalachia, model_df[[var]]))$p.value,
      error = function(e) NA_real_
    )

    Characteristic = label,
    Overall = fmt_bin_pct(overall),
    Non_Appalachia = fmt_bin_pct(non),
    Appalachia = fmt_bin_pct(app),

  )
}
```

## Table 3: Logistic regression — clinical characteristics

```{r table2-clinical}
library(dplyr)
library(sandwich)
library(lmtest)
library(broom)

# NOTE: `appalachia` is a county-level constant (every record in the same
# county has the same value), so a county random intercept (glmer with
# (1 | ResidenceFIPS)) is not identifiable for this outcome — it produces
# a degenerate fit with astronomical CIs (confirmed earlier in this
# project). glm() with county-clustered robust SEs is the correct approach
# for any model where `appalachia` is the outcome.

m_clinical <- glm(
  appalachia ~ age_grp + sex_f + race_eth_f + year_f +
    has_depression + has_bipolar + has_adhd + has_schizo + has_anyanxiety +
    combinedotherdiagnosis +MentalIllnessTreatment + DepressedMood_c +
    InCrisis + RelationshipProblem + SuicideAttemptHistory_c +
    SuicideThoughtHistory_c + LegalProblem + method_cat,
  data = model_df,
  family = binomial()
)

vc_clinical <- vcovCL(m_clinical, cluster = ~ ResidenceFIPS)

table3 <- tidy(coeftest(m_clinical, vcov. = vc_clinical)) %>%
  mutate(
    OR = exp(estimate),
    CI_low = exp(estimate - 1.96 * std.error),
    CI_high = exp(estimate + 1.96 * std.error)
  ) %>%
  filter(term != "(Intercept)") %>%
  transmute(
    term,
```

```{r}
  if (is.na(p)) return("")

make_cat_row <- function(label, var, level) {

  overall <- model_df[[var]]

  app <- model_df %>%
    filter(appalachia == 1) %>%
    pull(.data[[var]])

  non <- model_df %>%
    filter(appalachia == 0) %>%

  tibble(
    Characteristic = label,
    Overall = sprintf(
      "%s (%.1f%%)",

      100 * mean(overall == level, na.rm = TRUE)
    ),
```

```{r method-by-svi-theme}
    ResidenceFIPS  = factor(ResidenceFIPS),
    firearm_yes    = if_else(method_cat == "Firearm", 1L, 0L),
    hanging_yes    = if_else(method_cat == "Hanging/Suffocation", 1L, 0L),
    poisoning_yes  = if_else(method_cat == "Poisoning", 1L, 0L)
  )

fit_method <- function(outcome) {
  f <- as.formula(
    paste0(outcome, " ~ appalachia + age_grp + sex_f + race_f + eth_f + year_f + ",
           "RPL_THEME1 + RPL_THEME2 + RPL_THEME3 + RPL_THEME4 + ",
           "(1 | ResidenceState/ResidenceFIPS)")
  )
  glmer(f, data = df_m, family = binomial(), control = glmerControl(optimizer = "bobyqa"))
}

m_firearm   <- fit_method("firearm_yes")
m_hanging   <- fit_method("hanging_yes")
m_poisoning <- fit_method("poisoning_yes")

extract_or <- function(m, label) {
  broom.mixed::tidy(m, effects = "fixed", conf.int = TRUE) %>%
    mutate(model = label, OR = exp(estimate), CI_low = exp(conf.low), CI_high = exp(conf.high)) %>%
    select(model, term, OR, CI_low, CI_high, p.value)
}
```
